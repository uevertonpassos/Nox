cmake_minimum_required(VERSION 3.22)
project(nox VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network Svg)
add_executable(nox WIN32 src/main.cpp src/window.cpp src/window.h src/system.cpp src/system.h src/installer.cpp src/installer.h src/language.cpp src/language.h src/theme.cpp src/theme.h resources/resources.qrc)
if(WIN32)
  enable_language(RC)
  target_sources(nox PRIVATE resources/nox.rc)
  target_include_directories(nox PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/resources")
endif()
target_link_libraries(nox PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg)
if(MSVC)
  target_compile_options(nox PRIVATE /W4 /utf-8)
else()
  target_compile_options(nox PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(GNUInstallDirs)
install(TARGETS nox RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.noxdev.Nox.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/nox.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
foreach(size 48 128 256 512)
  install(FILES resources/icons/${size}x${size}/nox.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps)
endforeach()
install(FILES resources/logos/SOURCES.md resources/logos/DEVICON-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/nox/logos)
install(FILES THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/nox)
install(DIRECTORY resources/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/nox/licenses)
if(NOT WIN32)
  install(PROGRAMS packaging/install.sh DESTINATION .)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(system_test tests/system_test.cpp src/system.cpp src/system.h src/language.cpp src/language.h resources/resources.qrc)
  target_link_libraries(system_test PRIVATE Qt6::Widgets Qt6::Test)
  if(MSVC)
    target_compile_options(system_test PRIVATE /utf-8)
  endif()
  add_test(NAME system COMMAND system_test)
  add_executable(window_test tests/window_test.cpp src/window.cpp src/window.h src/installer.cpp src/installer.h src/system.cpp src/system.h src/language.cpp src/language.h src/theme.cpp src/theme.h resources/resources.qrc)
  target_link_libraries(window_test PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg Qt6::Test)
  if(MSVC)
    target_compile_options(window_test PRIVATE /utf-8)
  endif()
  add_test(NAME window COMMAND window_test)
  set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_test(NAME desktop-smoke COMMAND nox --smoke-test)
  set_tests_properties(desktop-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
set(CPACK_PACKAGE_NAME "Nox")
set(CPACK_PACKAGE_VENDOR "Nox")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Developer tools installer")
if(WIN32)
  set(CPACK_GENERATOR "NSIS")
  set(CPACK_NSIS_DISPLAY_NAME "Nox")
  set(CPACK_NSIS_PACKAGE_NAME "Nox")
  set(CPACK_NSIS_EXECUTABLES_DIRECTORY "bin")
  set(CPACK_PACKAGE_EXECUTABLES "nox;Nox")
  set(CPACK_NSIS_INSTALLED_ICON_NAME "bin\\\\nox.exe")
  set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
  set(CPACK_NSIS_MUI_ICON "${CMAKE_CURRENT_SOURCE_DIR}/resources/nox.ico")
  set(CPACK_NSIS_MUI_UNIICON "${CMAKE_CURRENT_SOURCE_DIR}/resources/nox.ico")
  if(DEFINED NOX_DEPLOY_DIR)
    install(DIRECTORY "${NOX_DEPLOY_DIR}/" DESTINATION bin)
  endif()
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
